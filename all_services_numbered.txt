    14	        string ffmpegPath,
    15	        IReadOnlyList<VideoEntry> entries,
    16	        Action<string> log,
    17	        CancellationToken cancellationToken)
    18	    {
    19	        foreach (var entry in entries)
    20	        {
    21	            cancellationToken.ThrowIfCancellationRequested();
    22	
    23	            if (entry.IsDirectMp4)
    24	            {
    25	                entry.CacheState = "Not needed";
    26	                continue;
    27	            }
    28	
    29	            if (string.IsNullOrWhiteSpace(entry.PlaybackCachePath))
    30	                throw new InvalidOperationException($"Playback cache path is missing for {entry.FileName}");
    31	
    32	            if (NeedsRebuild(entry.FullPath, entry.PlaybackCachePath))
    33	            {
    34	                entry.CacheState = "Converting playback";
    35	                log($"Creating playable MP4: {entry.FileName}");
    36	                await ConvertToPlaybackMp4Async(ffmpegPath, entry.FullPath, entry.PlaybackCachePath, cancellationToken);
    37	            }
    38	            else
    39	            {
    40	                log($"Using playable MP4 cache: {Path.GetFileName(entry.PlaybackCachePath)}");
    41	            }
    42	
    43	            if (!string.IsNullOrWhiteSpace(entry.TranslationCachePath))
    44	            {
    45	                if (NeedsRebuild(entry.FullPath, entry.TranslationCachePath))
    46	                {
    47	                    entry.CacheState = "Building translation proxy";
    48	                    log($"Creating proxy MP4 for translation: {entry.FileName}");
    49	                    await ConvertToProxyMp4Async(ffmpegPath, entry.FullPath, entry.TranslationCachePath, cancellationToken);
    50	                }
    51	                else
    52	                {
    53	                    log($"Using translation proxy cache: {Path.GetFileName(entry.TranslationCachePath)}");
    54	                }
    55	            }
    56	
    57	            entry.CacheState = "Ready";
    58	        }
    59	    }

   104	    private static async Task RunFfmpegAsync(string ffmpegPath, IReadOnlyList<string> arguments, string outputPath, CancellationToken cancellationToken)
   105	    {
   106	        Directory.CreateDirectory(Path.GetDirectoryName(outputPath)!);
   107	
   108	        var psi = new ProcessStartInfo
   109	        {
   110	            FileName = ffmpegPath,
   111	            UseShellExecute = false,
   112	            RedirectStandardOutput = true,
   113	            RedirectStandardError = true,
   114	            CreateNoWindow = true,
   115	        };
   116	
   117	        foreach (var arg in arguments)
   118	            psi.ArgumentList.Add(arg);
   119	
   120	        using var process = new Process { StartInfo = psi };
   121	        process.Start();
   122	
   123	        using var reg = cancellationToken.Register(() =>
   124	        {
   125	            try
   126	            {
   127	                if (!process.HasExited)
   128	                    process.Kill(entireProcessTree: true);
   129	            }
   130	            catch
   131	            {
   132	            }
   133	        });
   134	
   135	        var stdoutTask = process.StandardOutput.ReadToEndAsync();
   136	        var stderrTask = process.StandardError.ReadToEndAsync();
   137	        await process.WaitForExitAsync(cancellationToken);
   138	
   139	        var stdout = await stdoutTask;
   140	        var stderr = await stderrTask;
   141	
   142	        if (process.ExitCode != 0 || !File.Exists(outputPath))
   143	        {
   144	            throw new InvalidOperationException(
   145	                $"ffmpeg failed.\nOutput: {outputPath}\n\nSTDOUT:\n{stdout}\n\nSTDERR:\n{stderr}");
   146	        }
   147	    }
   148	
   149	    private static bool NeedsRebuild(string sourcePath, string outputPath)
   150	    {
   151	        if (!File.Exists(outputPath))
   152	            return true;
   153	
   154	        var source = new FileInfo(sourcePath);
   155	        var output = new FileInfo(outputPath);
   156	
   157	        return output.Length <= 0 || output.LastWriteTimeUtc < source.LastWriteTimeUtc;
   158	    }

    10	    {
    11	        TcpListener? listener = null;
    12	        try
    13	        {
    14	            listener = new TcpListener(IPAddress.Loopback, port);
    15	            listener.Start();
    16	            return true;
    17	        }
    18	        catch
    19	        {
    20	            return false;
    21	        }
    22	        finally
    23	        {

    26	    }

    28	    public static int GetFreeTcpPort(int preferredPort = 8000, int maxAttempts = 50)
    29	    {
    30	        if (TestPortBindable(preferredPort))
    31	            return preferredPort;
    32	
    33	        for (var port = preferredPort + 1; port < preferredPort + maxAttempts; port++)
    34	        {
    35	            if (TestPortBindable(port))
    36	                return port;
    37	        }
    38	
    39	        using var listener = new TcpListener(IPAddress.Loopback, 0);
    40	        listener.Start();
    41	        return ((IPEndPoint)listener.LocalEndpoint).Port;
    42	    }

    45	    {
    46	        using var client = new TcpClient();
    47	        try
    48	        {
    49	            var asyncResult = client.BeginConnect(hostName, portNumber, null, null);
    50	            var ok = asyncResult.AsyncWaitHandle.WaitOne(timeoutMs, false);
    51	            if (!ok)
    52	                return false;
    53	
    54	            client.EndConnect(asyncResult);

    70	            tools.CloudflaredPath,

    72	            workspace.RootFolder,

    74	            _cloudflaredStderrFile);

    76	                TranslationUrl = isDirectMp4 ? originalUrl : (proxyTranslationEnabled ? proxyUrl : playbackUrl),
    77	                PlaybackCachePath = isDirectMp4 ? null : playbackCachePath,
    78	                TranslationCachePath = isDirectMp4 ? null : (proxyTranslationEnabled ? proxyCachePath : null),
    79	                CacheState = GetInitialCacheState(file, playbackCachePath, isDirectMp4, proxyTranslationEnabled, proxyCachePath)
    80	            };
    81	
    82	            Entries.Add(entry);
    83	        }
    84	    }
    85	
    86	    private static string GetInitialCacheState(string sourcePath, string playbackCachePath, bool isDirectMp4, bool proxyEnabled, string proxyCachePath)
    87	    {
    88	        if (isDirectMp4)
    89	            return "Not needed";
    90	
    91	        var sourceInfo = new FileInfo(sourcePath);
    92	        var playbackReady = File.Exists(playbackCachePath) && new FileInfo(playbackCachePath).Length > 0 && new FileInfo(playbackCachePath).LastWriteTimeUtc >= sourceInfo.LastWriteTimeUtc;
    93	        var proxyReady = !proxyEnabled || (File.Exists(proxyCachePath) && new FileInfo(proxyCachePath).Length > 0 && new FileInfo(proxyCachePath).LastWriteTimeUtc >= sourceInfo.LastWriteTimeUtc);
    94	
    95	        return playbackReady && proxyReady ? "Ready" : "Pending";
    96	    }

    98	    public static string NewSlug(string text)
    99	    {
   100	        var slug = text.ToLowerInvariant();
   101	        slug = Regex.Replace(slug, "[^a-z0-9]+", "-");
   102	        slug = slug.Trim('-');
   103	        return string.IsNullOrWhiteSpace(slug) ? "video" : slug;
   104	    }

   106	    public static string EncodeUrlPath(string path)
   107	    {
   108	        var normalized = path.Replace('\\', '/');
   109	        var segments = normalized.Split('/', StringSplitOptions.None);
   110	        return string.Join("/", segments.Select(Uri.EscapeDataString));
   111	    }
